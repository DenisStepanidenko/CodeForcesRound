    private static String solve(int n, int m) {
        StringBuilder ans = new StringBuilder();

        if (n > m) {
            ans.append("-1");
            return ans.toString();
        } else {
            while (true) {
                if (m % n == 0) {
                    int number = m / n;
                    for (int j = 0; j < n; j++) {
                        ans.append(number).append(" ");
                    }
                    return ans.toString();
                }
                ans.append(1).append(" ");
                m--;
                n--;
            }
        }
    }
